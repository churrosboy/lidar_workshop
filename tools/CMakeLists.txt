cmake_minimum_required(VERSION 3.16)
project(gl5_lab LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(SOSLAB_SDK_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../SOSLAB_SDK/_archive_" CACHE PATH "Built SDK")
include(../cmake/SoslabSdk.cmake)
find_package(Threads REQUIRED)
add_executable(gl5_receive gl5_receive.cpp)
target_link_libraries(gl5_receive PRIVATE soslab_sdk Threads::Threads)
target_compile_options(gl5_receive PRIVATE -Wall -Wextra -Wpedantic)
