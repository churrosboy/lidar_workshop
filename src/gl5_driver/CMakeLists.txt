cmake_minimum_required(VERSION 3.16)
project(gl5_driver)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(sensor_msgs REQUIRED)
get_filename_component(LAB_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
set(SOSLAB_SDK_ROOT "${LAB_ROOT}/SOSLAB_SDK/_archive_" CACHE PATH "Built SOSLAB SDK")
include("${LAB_ROOT}/cmake/SoslabSdk.cmake")
add_library(gl5_conversion STATIC src/conversion.cpp)
target_include_directories(gl5_conversion PUBLIC include)
target_link_libraries(gl5_conversion soslab_sdk)
ament_target_dependencies(gl5_conversion sensor_msgs)
add_executable(gl5_node src/gl5_node.cpp)
target_link_libraries(gl5_node gl5_conversion)
ament_target_dependencies(gl5_node rclcpp sensor_msgs)
target_compile_options(gl5_node PRIVATE -Wall -Wextra -Wpedantic)
set_target_properties(gl5_node PROPERTIES INSTALL_RPATH "$ORIGIN/..")
install(TARGETS gl5_node DESTINATION lib/${PROJECT_NAME})
install(FILES "${SOSLAB_LIBRARY}" DESTINATION lib)
install(DIRECTORY config DESTINATION share/${PROJECT_NAME})
ament_package()
