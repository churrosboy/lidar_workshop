cmake_minimum_required(VERSION 3.16)
project(gl5_driver)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(rviz_common REQUIRED)
find_package(pluginlib REQUIRED)
find_package(std_srvs REQUIRED)
find_package(std_msgs REQUIRED)
find_package(Qt5 REQUIRED COMPONENTS Widgets)
add_library(gl5_roi_panel SHARED src/roi_panel.cpp)
target_link_libraries(gl5_roi_panel Qt5::Widgets)
ament_target_dependencies(gl5_roi_panel rviz_common pluginlib rclcpp std_srvs std_msgs)
pluginlib_export_plugin_description_file(rviz_common roi_panel_plugin.xml)
install(TARGETS gl5_roi_panel LIBRARY DESTINATION lib)
get_filename_component(LAB_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
set(SOSLAB_SDK_ROOT "${LAB_ROOT}/SOSLAB_SDK/_archive_" CACHE PATH "Built SOSLAB SDK")
include("${LAB_ROOT}/cmake/SoslabSdk.cmake")
add_library(gl5_conversion STATIC src/conversion.cpp)
target_include_directories(gl5_conversion PUBLIC include)
target_link_libraries(gl5_conversion soslab_sdk)
ament_target_dependencies(gl5_conversion sensor_msgs)
add_executable(gl5_node src/gl5_node.cpp)
target_link_libraries(gl5_node gl5_conversion)
ament_target_dependencies(gl5_node rclcpp sensor_msgs)
target_compile_options(gl5_node PRIVATE -Wall -Wextra -Wpedantic)
set_target_properties(gl5_node PROPERTIES INSTALL_RPATH "$ORIGIN/..")
install(TARGETS gl5_node DESTINATION lib/${PROJECT_NAME})
install(FILES "${SOSLAB_LIBRARY}" DESTINATION lib)
install(DIRECTORY launch config rviz DESTINATION share/${PROJECT_NAME})
install(PROGRAMS python/gl5_obstacle_node.py DESTINATION lib/${PROJECT_NAME})
install(FILES python/roi_geometry.py python/roi_tracking.py DESTINATION lib/${PROJECT_NAME})
if(BUILD_TESTING)
  find_package(Python3 REQUIRED COMPONENTS Interpreter)
  add_test(NAME test_roi_geometry COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/test/test_roi_geometry.py)
  add_test(NAME test_roi_tracking COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/test/test_roi_tracking.py)
  find_package(ament_cmake_gtest REQUIRED)
  ament_add_gtest(test_conversion test/test_conversion.cpp)
  target_link_libraries(test_conversion gl5_conversion)
endif()
ament_package()
